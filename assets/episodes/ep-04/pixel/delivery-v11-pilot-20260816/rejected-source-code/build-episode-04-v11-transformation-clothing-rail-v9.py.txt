def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def ease(value: float) -> float:
    value = min(1.0, max(0.0, value))
    return value * value * (3.0 - 2.0 * value)


def clean_cutout(path: Path) -> Image.Image:
    cutout = Image.open(path).convert("RGBA")
    alpha = cutout.getchannel("A")
    # Apple Vision retains a coloured source-stage rim. A modest erosion removes
    # the rim while preserving the already-approved face, clothes and silhouette.
    alpha = alpha.filter(ImageFilter.MinFilter(9)).filter(ImageFilter.GaussianBlur(0.55))
    cutout.putalpha(alpha)
    return cutout

                  (x + 101, y + 28), (x + 81, y + 62), (x + 61, y + 28)]
        draw.polygon(points, fill=colour + (255,), outline=ink)
        draw.line(points + [points[0]], fill=ink, width=9, joint="curve")
        draw.polygon([(x + 81, y + 62), (x + 58, y + 135), (x + 81, y + 116)], fill=shadow, outline=ink)
        draw.polygon([(x + 81, y + 62), (x + 104, y + 135), (x + 81, y + 116)], fill=shadow, outline=ink)
        draw.line((x + 81, y + 116, x + 81, y + 300), fill=ink, width=7)
        if plaid:

            for offset in (105, 155, 210, 260):

    elif kind == "skirt":

            for offset in (42, 72, 102, 132):

def title_tag(frame: Image.Image, opacity: float) -> Image.Image:
    layer = Image.new("RGBA", frame.size, (0, 0, 0, 0))
    draw = ImageDraw.Draw(layer)
    alpha = round(255 * min(1.0, max(0.0, opacity)))
    draw.polygon([(410, 365), (1510, 365), (1450, 720), (470, 720)], fill=(241, 223, 166, alpha), outline=(13, 17, 30, alpha), width=14)
    draw.ellipse((450, 400, 500, 450), fill=(15, 18, 31, alpha))
    draw.line((475, 425, 300, 220), fill=(226, 178, 77, alpha), width=16)
    kicker = ImageFont.truetype(str(FONT), 72)
    main = ImageFont.truetype(str(FONT), 155)
    draw.text((980, 463), "NOW ENTERING", font=kicker, anchor="mm", fill=(70, 46, 60, alpha))
    draw.text((982, 605), "SUNNYVAiLE", font=main, anchor="mm", fill=(217, 167, 35, alpha), stroke_width=10, stroke_fill=(13, 17, 30, alpha))
    return Image.alpha_composite(frame.convert("RGBA"), layer)


def slide(base: Image.Image, overlay: Image.Image, progress: float, entering: bool) -> Image.Image:
    progress = ease(progress)
    x = round(1920 * (1.0 - progress)) if entering else round(-1920 * progress)
    frame = base.convert("RGBA")
    frame.alpha_composite(overlay, (x, 0))
    return frame


def page_wipe(old: Image.Image, new: Image.Image, progress: float) -> Image.Image:
    """Replace old with new under a diagonal black-and-gold comic gutter."""
    progress = ease(progress)
    edge = round(-340 + (1920 + 680) * progress)
    mask = Image.new("L", (1920, 1080), 0)
    draw = ImageDraw.Draw(mask)
    draw.polygon([(edge - 340, 0), (1920, 0), (1920, 1080), (edge + 340, 1080)], fill=255)
    frame = Image.composite(new.convert("RGBA"), old.convert("RGBA"), mask)
    gutter = Image.new("RGBA", (1920, 1080), (0, 0, 0, 0))
    gutter_draw = ImageDraw.Draw(gutter)
    gutter_draw.line((edge - 340, 0, edge + 340, 1080), fill=(10, 12, 20, 255), width=46)
    gutter_draw.line((edge - 323, 0, edge + 357, 1080), fill=(224, 175, 74, 255), width=10)
    return Image.alpha_composite(frame, gutter)


def main() -> None:
    for path, expected in SOURCE_HASHES.items():
        observed = sha256(path)
        if observed != expected:
            raise SystemExit(f"SOURCE DRIFT {path}: {observed} != {expected}")
    if not PLAID_CUTOUT.exists():
        raise SystemExit(f"MISSING APPROVED-PIXEL CUTOUT {PLAID_CUTOUT}")
    before = Image.open(BEFORE).convert("RGBA")
    after = Image.open(AFTER).convert("RGBA")
    heroine = clean_cutout(PLAID_CUTOUT)
    corporate = rail_plate("corporate")
    sunny = rail_plate("sunny")
    # Keep the front-facing identity proof in the wardrobe environment. The
    # separate exact travel shot follows under a page wipe, so two Heroine poses
    # can never appear in the same frame.
    reveal = sunny.copy()
    reveal.alpha_composite(heroine, (0, 0))

    with tempfile.TemporaryDirectory(prefix="ep04-clothing-rail-v8-") as temp_name:
        temp = Path(temp_name)
        for index in range(FRAMES):
            if index < 24:
                frame = before
            elif index < 60:
                frame = slide(before, corporate, (index - 24) / 35, True)
            elif index < 78:
                frame = corporate
            elif index < 96:
                frame = title_tag(corporate, (index - 78) / 17)
            elif index < 126:
                blend = ease((index - 96) / 29)
                frame = Image.blend(title_tag(corporate, 1.0).convert("RGB"), title_tag(sunny, 1.0).convert("RGB"), blend).convert("RGBA")
            elif index < 144:
                frame = title_tag(sunny, (143 - index) / 17)
            elif index < 180:
                frame = slide(reveal, sunny, (index - 144) / 35, False)
            elif index < 210:
                frame = reveal
            elif index < 228:
                frame = page_wipe(reveal, after, (index - 210) / 17)
            else:
                frame = after
            frame.convert("RGB").save(temp / f"frame-{index:04d}.png")
        subprocess.run([
            str(FFMPEG), "-hide_banner", "-loglevel", "error", "-y", "-framerate", str(FPS),
            "-i", str(temp / "frame-%04d.png"), "-frames:v", str(FRAMES), "-an", "-c:v", "libx264",
            "-preset", "slow", "-crf", "16", "-pix_fmt", "yuv420p", "-movflags", "+faststart", str(OUTPUT),
        ], check=True)
    decoded, seconds = imageio_ffmpeg.count_frames_and_secs(str(OUTPUT))
    if decoded != FRAMES or abs(seconds - FRAMES / FPS) > 1 / FPS:
        raise SystemExit(f"CLOCK FAIL {decoded=} {seconds=}")
    subprocess.run([str(FFMPEG), "-hide_banner", "-loglevel", "error", "-y", "-i", str(OUTPUT), "-vf", "fps=5,scale=480:270,tile=8x5", "-frames:v", "1", str(CONTACT)], check=True)
    subprocess.run([str(FFMPEG), "-hide_banner", "-loglevel", "error", "-y", "-i", str(OUTPUT), "-vf", "select='between(n,18,225)',scale=320:180,tile=12x18", "-vsync", "0", "-frames:v", "1", str(SWITCH)], check=True)
    record = {
        "status": "BUILT_LOCALLY_INTERNAL_IDENTITY_AND_MECHANISM_PILOT_NOT_RELEASE_AUTHORITY",
        "job": "Make the corporate-to-SUNNYVAiLE outfit change and Heroine identity unmistakable without regenerating her.",
        "mechanism": "Exact corporate close-up; literal clothing rail changes from corporate navy to yellow-plaid Rewind Era wardrobe under a deterministic NOW ENTERING SUNNYVAiLE garment tag; front-facing approved Heroine reveal inside that wardrobe; inked page wipe into the exact rear-view LUMINAiRY travel shot.",
        "sources": [
            {"path": str(path), "sha256": expected} for path, expected in SOURCE_HASHES.items()
        ] + [{"path": str(PLAID_CUTOUT), "sha256": sha256(PLAID_CUTOUT), "use": "approved p4 pixels isolated with Apple Vision; no character generation"}],
        "font": {"path": str(FONT), "sha256": sha256(FONT)},
        "output": {"path": str(OUTPUT), "sha256": sha256(OUTPUT), "frames": decoded, "seconds": seconds},
        "contactSheet": {"path": str(CONTACT), "sha256": sha256(CONTACT)},
        "switchContactSheet": {"path": str(SWITCH), "sha256": sha256(SWITCH)},
        "constraints": [
            "no new character generation", "no substitute figure", "front-facing identity proof before rear-view travel shot",
            "no fake closet doors", "no stage background", "no cloud", "no flash", "no generated text", "silent pilot only"
        ],
    }
    MANIFEST.write_text(json.dumps(record, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(record, indent=2))


if __name__ == "__main__":
    main()
